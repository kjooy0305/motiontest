function esc(s){ return s.replace(/&/g,'&').replace(//g,'>'); }
function highlight(code){
  const slots=[]; const st=h=>{slots.push(h);return`\x00${slots.length-1}\x00`;};
  let h=esc(code);
  h=h.replace(/"""[\s\S]*?"""|'''[\s\S]*?'''/g,m=>st(`${m}`));
  h=h.replace(/`[^`]*`/g,m=>st(`${m}`));
  h=h.replace(/"(?:[^"\\]|\\.)*"|'(?:[^'\\]|\\.)*'/g,m=>st(`${m}`));
  h=h.replace(/\/\/[^\n]*/g,m=>st(`${m}`));
  h=h.replace(/#(?!include)[^\n]*/g,m=>st(`${m}`));
  h=h.replace(/\/\*[\s\S]*?\*\//g,m=>st(`${m}`));
  h=h.replace(/--[^\n]*/g,m=>st(`${m}`));
  h=h.replace(/@[\w.]+/g,m=>st(`${m}`));
  h=h.replace(/\b([A-Za-z_]\w*)\b/g,(m,w)=>{
    if(KWS.has(w)) return st(`${w}`);
    if(BIS.has(w)) return st(`${w}`);
    return m;
  });
  h=h.replace(/\b(\d[\d_]*\.?\d*([eE][+-]?\d+)?)\b/g,m=>st(`${m}`));
  return h.replace(/\x00(\d+)\x00/g,(_,i)=>slots[+i]);
}

function buildNav(q=''){
  const nav=document.getElementById('nav');

      return d.cat===cat;
    });
